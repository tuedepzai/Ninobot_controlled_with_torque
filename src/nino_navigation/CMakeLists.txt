cmake_minimum_required(VERSION 3.8)
project(nino_navigation)
find_package(ament_cmake REQUIRED)
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/episode_manager.py
  DESTINATION lib/${PROJECT_NAME} RENAME episode_manager)
install(FILES scripts/path_deviation.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/odom_warmup.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(path_deviation test/test_path_deviation.py)
  ament_add_pytest_test(odom_warmup test/test_odom_warmup.py)
  ament_add_pytest_test(episode_manager test/test_episode_manager.py
    TIMEOUT 90)
endif()
ament_package()
